# download, ~1 GB on disk the first time it installs.
# freeto's PardisoSolver uses private pypardiso internals -> keep the <0.5 pin
# until fe.py is updated to track a newer pypardiso release.
pypardiso>=0.4.6,<0.5; (sys_platform == "win32" or sys_platform == "linux") and platform_machine in "x86_64 AMD64"
